#  查询数据表的命令

## PostgreSQL（技能相关表）

# 工具语义表（看描述 + schema + slots）
docker exec -it postgres psql -U root -d postgres -c "SELECT skill_name, tool_name, left(tool_description,120) AS tool_desc_prefix, left(input_schema,120) AS input_schema_prefix, left(input_slots,120) AS input_slots_prefix, updated_at FROM mcp_tool_semantic ORDER BY updated_at DESC LIMIT 100;"

# 工具向量表（维度 + 热度）
docker exec -it postgres psql -U root -d postgres -c "SELECT skill_name, tool_name, recent_7d_count, heat_weight, jsonb_array_length(normalized_vector) AS vector_dim, updated_at FROM mcp_tool_vector ORDER BY updated_at DESC LIMIT 100;"

# 技能语义表（看 intent/tags）
docker exec -it postgres psql -U root -d postgres -c "SELECT skill_name, intent, tags, left(skill_description,120) AS skill_desc_prefix, updated_at FROM skill_semantic_snapshot ORDER BY updated_at DESC LIMIT 50;"

# 技能向量表（看 intent/tags + 向量维度）
docker exec -it postgres psql -U root -d postgres -c "SELECT skill_name, intent, tags, jsonb_array_length(skill_vector) AS vector_dim, updated_at FROM skill_vector_snapshot ORDER BY updated_at DESC LIMIT 50;"

截断技能表

docker exec -it postgres psql -U root -d postgres -c "TRUNCATE TABLE skill_vector_snapshot, skill_semantic_snapshot RESTART IDENTITY;"


截断工具表
docker exec -it postgres psql -U root -d postgres -c "TRUNCATE TABLE mcp_tool_semantic, mcp_tool_vector RESTART IDENTITY;"

## 查看 xxl-job 容器日志（tail）

### 查看最近 100 行日志
```bash
docker logs --tail 100 xxl-job-admin
```

### 持续跟踪最新日志
```bash
docker logs -f --tail 100 xxl-job-admin
```

### 只看错误相关日志
```bash
docker logs --tail 500 xxl-job-admin 2>&1 | grep -Ei "error|exception|fail|timeout|refused"
```
